-- 1. ACTUALIZAR public.notes (Autoría UUID, audiencia y marca de contenido demo)
ALTER TABLE public.notes
  ADD COLUMN IF NOT EXISTS author_id UUID REFERENCES public.users(id) ON DELETE CASCADE,
  ADD COLUMN IF NOT EXISTS book_id INT REFERENCES public.books(id) ON DELETE SET NULL,
  ADD COLUMN IF NOT EXISTS note_type VARCHAR(20) DEFAULT 'quote'
    CHECK (note_type IN ('note', 'quote', 'photo', 'video')),
  ADD COLUMN IF NOT EXISTS audience VARCHAR(20) DEFAULT 'public'
    CHECK (audience IN ('public', 'followers', 'private')),
  ADD COLUMN IF NOT EXISTS page_number TEXT,
  ADD COLUMN IF NOT EXISTS media_url TEXT,
  ADD COLUMN IF NOT EXISTS is_demo BOOLEAN DEFAULT FALSE;

UPDATE public.notes SET is_demo = TRUE WHERE author_id IS NULL;

ALTER TABLE public.notes ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Lectura pública de notas" ON public.notes;
DROP POLICY IF EXISTS "Permitir insercion publica de notas" ON public.notes;
DROP POLICY IF EXISTS "Permitir actualizacion publica de notas" ON public.notes;

CREATE POLICY "Leer notas permitidas" ON public.notes
  FOR SELECT USING (
    audience = 'public'
    OR author_id = auth.uid()
  );

CREATE POLICY "Crear notas propias" ON public.notes
  FOR INSERT WITH CHECK (
    auth.uid() IS NOT NULL AND author_id = auth.uid()
  );

CREATE POLICY "Editar notas propias" ON public.notes
  FOR UPDATE USING (author_id = auth.uid())
  WITH CHECK (author_id = auth.uid());

CREATE POLICY "Borrar notas propias" ON public.notes
  FOR DELETE USING (author_id = auth.uid());

  note_id UUID NOT NULL REFERENCES public.notes(id) ON DELETE CASCADE,
  created_at TIMESTAMPTZ DEFAULT NOW(),

);

ALTER TABLE public.note_likes ENABLE ROW LEVEL SECURITY;

  FOR ALL USING (auth.uid() = user_id) WITH CHECK (auth.uid() = user_id);

-- Trigger: recalcula likes_count en servidor sin abrir UPDATE en public.notes
CREATE OR REPLACE FUNCTION public.sync_note_likes_count()
RETURNS TRIGGER LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$
DECLARE
  target_note UUID;
BEGIN
  target_note := COALESCE(NEW.note_id, OLD.note_id);
  UPDATE public.notes
  SET likes_count = (SELECT COUNT(*) FROM public.note_likes WHERE note_id = target_note)
  WHERE id = target_note;
  RETURN NULL;
END;
$$;

DROP TRIGGER IF EXISTS trg_sync_note_likes ON public.note_likes;
CREATE TRIGGER trg_sync_note_likes
  AFTER INSERT OR DELETE ON public.note_likes
  FOR EACH ROW EXECUTE FUNCTION public.sync_note_likes_count();

-- 3. BIBLIOTECA PERSONAL, VALORACIONES Y RESEÑAS (public.user_books)
CREATE TABLE IF NOT EXISTS public.user_books (
  user_id UUID NOT NULL REFERENCES public.users(id) ON DELETE CASCADE,
  book_id INT NOT NULL REFERENCES public.books(id) ON DELETE CASCADE,
  status VARCHAR(20) NOT NULL DEFAULT 'want_to_read'
    CHECK (status IN ('want_to_read', 'reading', 'read', 'abandoned')),
  current_page INT DEFAULT 0 CHECK (current_page >= 0),
  rating INT CHECK (rating IS NULL OR (rating BETWEEN 1 AND 10)),
  review TEXT,
  review_audience VARCHAR(20) DEFAULT 'public'
    CHECK (review_audience IN ('public', 'community', 'private')),
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW(),
  PRIMARY KEY (user_id, book_id)
);

-- Migrar votos previos de book_ratings que ya correspondan a un UUID real
INSERT INTO public.user_books (user_id, book_id, status, rating)
SELECT u.id, br.book_id, 'read', br.rating
FROM public.book_ratings br
JOIN public.users u ON br.user_id = u.id::text
JOIN public.books b ON br.book_id = b.id
ON CONFLICT (user_id, book_id) DO UPDATE SET rating = EXCLUDED.rating;

-- Cerrar política abierta de la tabla heredada book_ratings
ALTER TABLE public.book_ratings ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Permitir insertar y actualizar ratings" ON public.book_ratings;
CREATE POLICY "Solo propietario gestiona book_ratings legacy" ON public.book_ratings
  FOR ALL USING (user_id = auth.uid()::text) WITH CHECK (user_id = auth.uid()::text);

-- RLS estricto en user_books (cada usuario gestiona su biblioteca)
ALTER TABLE public.user_books ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Usuario gestiona su biblioteca" ON public.user_books;
CREATE POLICY "Usuario gestiona su biblioteca" ON public.user_books
  FOR ALL USING (auth.uid() = user_id) WITH CHECK (auth.uid() = user_id);

-- Vista pública segura de reseñas y valoraciones comunitarias
CREATE OR REPLACE VIEW public.public_book_reviews AS
SELECT
  ub.book_id,
  ub.user_id,
  u.display_name,
  u.username,
  u.avatar_url,
  ub.rating,
  ub.review,
  ub.updated_at
FROM public.user_books ub
JOIN public.users u ON u.id = ub.user_id
WHERE (ub.rating IS NOT NULL OR NULLIF(trim(ub.review), '') IS NOT NULL)
  AND (
    ub.review_audience = 'public'
    OR (ub.review_audience = 'community' AND auth.uid() IS NOT NULL)
    OR ub.user_id = auth.uid()
  );

GRANT SELECT ON public.public_book_reviews TO anon, authenticated;

-- Trigger: recalcular media y recuento real en public.books al puntuar/editar/borrar
CREATE OR REPLACE FUNCTION public.sync_book_rating_stats()
RETURNS TRIGGER LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$
DECLARE
  target_book INT;
BEGIN
  target_book := COALESCE(NEW.book_id, OLD.book_id);
  UPDATE public.books
  SET
    votes_count = (SELECT COUNT(*) FROM public.user_books WHERE book_id = target_book AND rating IS NOT NULL),
    rating_avg = COALESCE(
      (SELECT ROUND(AVG(rating)::numeric, 1) FROM public.user_books WHERE book_id = target_book AND rating IS NOT NULL),
      0.0
    )
  WHERE id = target_book;
  RETURN NULL;
END;
$$;

DROP TRIGGER IF EXISTS trg_sync_book_stats ON public.user_books;
CREATE TRIGGER trg_sync_book_stats
  AFTER INSERT OR UPDATE OF rating OR DELETE ON public.user_books
  FOR EACH ROW EXECUTE FUNCTION public.sync_book_rating_stats();
